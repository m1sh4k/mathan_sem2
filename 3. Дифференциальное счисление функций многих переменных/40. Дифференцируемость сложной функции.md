## Формула для частных производных

$$
\begin{align*} \\
&F:=f(x_{1}(t),\dots,x_{m}(t))\\
&\frac{\delta F}{\delta t_{j}}(t^{0})=\sum_{i=1}^{m} \frac{\delta f}{\delta x_{j}}(a) \frac{\delta x_{i}}{\delta t_{j}}(t^{0})
\end{align*}
$$

## Условия
$j \in \mathbb{N}[1;m]$

1. $\varphi_{j}$ дифференцируемы в $t^{0}$
2. $f$ дифференцируема в точке $a$, где $a_{j}=x_{j}(t^{0})$

### Докво

- $t^{0}+\Delta t \in U(t^{0})$
- $a+\Delta x \in U(a)$
- $\rho=\sqrt{ (\Delta t_{1})^{2}+\dots+(\Delta t_{k})^{2} }$

#### Условие для $x_{j}(t)$
$$
\begin{align*}
\Delta x _{j}(t^{0},\Delta t)=\sum_{i=1}^{k} \frac{\partial x_{j}}{\partial t_{i}} (t^{0})\Delta t_{i}+\bar{\bar{o}}(\rho),\quad \rho\to0
\end{align*}
$$

#### Условие для $f(x)$
$$
\begin{align*}
\Delta f(a,\Delta x)=\sum_{i=1}^{m}\frac{ \partial f }{ \partial x_{i} } (a)\Delta x_{i}+\sum_{i=1}^{m}\alpha_{i}\Delta x_{i}
\end{align*}
$$

#### Подставляем приращения

$$
\begin{align*}
&\Delta f(a,\Delta x)=\frac{ \partial f }{ \partial x_{1} } (a)\left[ \frac{ \partial x_{1} }{ \partial t_{1} } (t^{0})\Delta t_{1}+\dots+\frac{ \partial x_{1} }{ \partial t_{k} } (t^{0})\Delta t_{k}+\bar{\bar{o}}(\rho) \right] +\dots+ \\
&+\frac{ \partial f }{ \partial x_{m} } (a)\left[ \frac{ \partial x_{m} }{ \partial t_{1} } (t^{0})\Delta t_{1}+\dots+\frac{ \partial x_{m} }{ \partial t_{k} } (t^{0})\Delta t_{k}+\bar{\bar{o}}(\rho) \right]+ \\
&+\alpha_{1}\left[ \frac{ \partial x_{1} }{ \partial t_{1} } (t^{0})\Delta t_{1} +\dots+\frac{ \partial x_{1} }{ \partial t_{k} } (t^{0})\Delta t_{k}+\bar{\bar{o}}(\rho)\right]+\dots+ \\
&+\alpha_{m}\left[ \frac{ \partial x_{m} }{ \partial t_{1} } (t^{0})\Delta t_{1} +\dots+\frac{ \partial x_{m} }{ \partial t_{k} } (t^{0})\Delta t_{k}+\bar{\bar{o}}(\rho)\right]
\end{align*}
$$

#### Группируем

$$
\begin{align*}
\Delta F(t^{0},\Delta t)=\sum_{j=1}^{k}\left( \sum_{i=1}^{m}\frac{ \partial f }{ \partial x_{i} } (a)\frac{ \partial x_{i} }{ \partial t_{j} }  \right) \Delta t_{j}+\bar{\bar{o}}(\rho)
\end{align*}
$$

#### Получаем

$$
\begin{align*}
\Delta F(t^{0})_{\Delta t}=\sum_{j=1}^{k}\frac{ \partial F }{ \partial t_{j} } (t^{0})\Delta t_{j}+\bar{\bar{o}}(\rho)
\end{align*}
$$
