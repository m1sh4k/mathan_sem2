
## Степень

$$
\begin{align*}
&\int a^{x }dx=\frac{a^{x }}{\ln a} +C\\
&\hline \\
&\left( \frac{a^{x}}{\ln a} \right) '= \frac{1}{\ln a}\left( e^{\ln a \cdot x} \right) '=a^{x}
\end{align*}
$$

## 1/син или кос^ 2

$$
\begin{align*}
\int \frac{dx}{\sin ^{2}x}=-\cot x+C  \\
\int \frac{dx}{\cos ^{2}x}=\tan x+C
\end{align*}
$$


## Арксинус

$$
\begin{align*}
&\int \frac{dx}{\sqrt{a^{2}-x^{2}}}=\arcsin \frac{x}{a}+C, |x|<|a| \\
&\hline \\
&\left(\arcsin \frac{x}{a}\right)'=\frac{1}{\sqrt{1-\left( \frac{x}{a} \right) ^{2} }}\cdot \left( \frac{x}{a} \right) '=\frac{1}{\cancel{ a } \sqrt{ \frac{a^{2}-x^{2}}{\cancel{ a^{2} }}}}=\frac{1}{\sqrt{ a^{2}-x^{2} }}
\end{align*}
$$

## Арктангенс

$$
\begin{align*}
&\int \frac{dx}{a^{2}+x^{2}}=\frac{1}{a}\arctan \frac{x}{a}+C \\
&\hline \\
& \left( \frac{1}{a}\arctan \frac{x}{a} \right) '=\frac{1}{a}\cdot \frac{1}{1+\left( \frac{x}{a} \right) ^{2}}\cdot \frac{1}{a}=\frac{1}{a^{2}+x^{2}}
\end{align*}
$$

## Высокий логарифм

$$
\begin{align*}
\int \frac{dx}{a^{2}-x^{2}}=\frac{1}{2a}\ln \left| \frac{a+x}{a-x} \right| +C \\
\end{align*}
$$

## Длинный логарифм

$$
\begin{align*}
\frac{dx}{\sqrt{ x^{2}\pm a^{2} }}=\ln |x+\sqrt{ x^{2}\pm a^{2} }|+C
\end{align*}
$$
