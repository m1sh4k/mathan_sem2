## Непрерывность

$$
\begin{align*}
F(x):=\int_{a}^{x}f(t)dt\implies F(x)\in C[a;b]
\end{align*}
$$
### Докво

$$
\begin{align*} \\
&x_{1}<x_{2}\in[a;b] \\
&f(x)\leq M\text{ на }[a;b]\text{ (интегрируема)}\\
&|F(x_{2})-F(x_{1})|=\left| \int_{a}^{x_{2}}f(t)dt-\int_{a}^{x_{1}}f(t)dt \right|=\left| \int_{x_{1}}^{x_{2}}f(t)dt \right| \leq  M \\
&\forall\varepsilon>0\exists\delta:=\frac{\varepsilon}{M}:\forall x_{1},x_{2}\in[a;b]\quad  |x_{2}-x_{1}|<\frac{\varepsilon}{M}\implies |F(x_{2})-F(x_{1})|<\varepsilon
\end{align*}
$$

##  По сторонам света

$$
\begin{align*}
&F(x)=\int_{a}^{x}f(t)dt \\
&a\leq x\leq b  \\
&x_{0}\in[a;b] 
\end{align*}
$$
1. $f\in C\{ x_{0}^{+} \}\implies F'_{x}(x_{0})=f(x_{0})$
2. $f\in C\{ x_{0}^{-} \}\implies F'_{-}(x_{0})=f(x_{0})$
3. $f\in C\{ x_{0} \}\implies F'(x_{0})=f(x_{0})$

### Докво

$$
\begin{align*}
&\varphi (x):=\left| \frac{F(x_{0}+x)-F(x_{0})}{x} -f(x_{0})\right|=\left| \frac{\int_{a}^{x_{0}+x}f(t)dt-\int_{a}^{x_{0}}f(t)dt}{x}-\frac{f(x_{0})}{x}\int_{x_{0}}^{x_{0}+x}1dt \right| = \\
&= \frac{\left| \int_{x_{0}}^{x_{0}+x}(f(t)-f(x_{0}))dt\right|_{}}{x}
\end{align*}
$$

$f \in C\{ x_{0}^{+} \}\implies\forall\varepsilon>0\exists\delta>0:\forall t\in[x_{0},x_{0}+\delta)\quad |f(t)-f(x_{0})|<\varepsilon$
$$
\begin{align*}
\left.   
\begin{aligned}
x \in[0,\delta) \\
t \in[x_{0},x_{0}+x]
\end{aligned}\right\} \implies   \left\{
\begin{aligned}
&t\in[x_{0},x_{0}+\delta] \\
&\varphi (x)= \frac{1}{x}\int_{x_{0}}^{x_{0}+x}|f(t)-f(x_{0})|dt\leq \frac{1}{x}\varepsilon \int_{x_{0}}^{x_{0}+x}1dt=\frac{1}{x}\cdot x\varepsilon=\varepsilon 
\end{aligned}\right.
\end{align*}
$$
